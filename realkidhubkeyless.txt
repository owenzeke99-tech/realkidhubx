-- Hàm lưu key và thời gian vào máy
local function saveKeyData(key)
    if writefile then
        local data = {
            key = key,
            timestamp = os.time()
        }
        pcall(function()
            writefile(SAVE_FILE_NAME, HttpService:JSONEncode(data))
        end)
    end
end

-- Hàm kiểm tra key đã lưu còn hạn không (dưới 24h)
local function isSavedKeyValid()
    if readfile and isfile and isfile(SAVE_FILE_NAME) then
        local success, result = pcall(function()
            local data = HttpService:JSONDecode(readfile(SAVE_FILE_NAME))
            if data and data.key == CORRECT_KEY and data.timestamp then
                local currentTime = os.time()
                if (currentTime - data.timestamp) < EXPIRE_TIME then
                    return true
                end
            end
        end)
        if success and result then
            return true
        end
    end
    return false
end

-- Hàm thực thi script chính
local function loadMainScript()
    local success, err = pcall(function()
        loadstring(game:HttpGet("https://raw.githubusercontent.com/realkidhub/realkid/refs/heads/main/main.lua"))()
    end)
    if not success then
        warn("Failed to load script: " .. tostring(err))
    end
end

    return -- Dừng không tạo giao diện nhập key nữa
end

-- Hover Effects for Buttons
local function addHoverEffect(button, normalBg, hoverBg)
    button.MouseEnter:Connect(function()
        TweenService:Create(button, TweenInfo.new(0.2), {BackgroundColor3 = hoverBg}):Play()
    end)
    button.MouseLeave:Connect(function()
        TweenService:Create(button, TweenInfo.new(0.2), {BackgroundColor3 = normalBg}):Play()
    end)
end

-- Smooth Shake Animation on Invalid Key
local isShaking = false
local function smoothShakeFrame()
    if isShaking then return end
    isShaking = true
    
    local originalPosition = UDim2.new(0.5, -210, 0.5, -140)
    local shakeDistance = 10
    local shakeSpeed = 25
    local duration = 0.5
    local elapsed = 0
    
    TweenService:Create(UIStroke, TweenInfo.new(0.2), {Color = Color3.fromRGB(255, 60, 60)}):Play()
    
    local connection
    connection = RunService.RenderStepped:Connect(function(dt)
        elapsed = elapsed + dt
        if elapsed < duration then
            local offset = math.sin(elapsed * shakeSpeed) * shakeDistance * (1 - (elapsed / duration))
            MainFrame.Position = originalPosition + UDim2.new(0, offset, 0, 0)
        else
            MainFrame.Position = originalPosition
            TweenService:Create(UIStroke, TweenInfo.new(0.4), {Color = Color3.fromRGB(0, 210, 255)}):Play()
            connection:Disconnect()
            isShaking = false
        end
    end)
end
